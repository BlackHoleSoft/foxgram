### 1.2 Config loader (`src/config.ts`)

```typescript
export interface ServerConfig {
  port: number;
  dbPath: string;
  jwtSecret: string;
  logLevel: string;
}

export function loadConfig(): ServerConfig
// - Загружает .env через dotenv
// - Возвращает конфиг с дефолтными значениями
// - PORT=3000, DB_PATH=./data/foxgram.db, LOG_LEVEL=debug
// - JWT_SECRET: бросить ошибку если не установлен
```

### 1.3 Logger service (`src/logger.ts`)

```typescript
export class Logger {
  private level: string;

  debug(...args: any[]): void
  info(...args: any[]): void
  warn(...args: any[]): void
  error(...args: any[]): void
}

export const logger = new Logger(config.logLevel)
```

- Поддержка уровней: debug, info, warn, error
- Формат: `[timestamp] [level] message`
- **Никогда не логировать пароли и секретные ключи**
