### 7.1 Создать `src/app.ts` — главный компонент приложения

```typescript
import { App as InkApp } from 'ink';
import React, { useState, useEffect } from 'react';
import { Foxgram } from 'foxgram-core';
import { Storage } from 'foxgram-core/storage';
import { LoginScreen } from './screens/login';
import { ContactListScreen } from './screens/contact-list';
import { ChatScreen } from './screens/chat';
import { AddContactScreen } from './screens/add-contact';
import { DeleteContactsScreen } from './screens/delete-contacts';

type Screen = 'login' | 'contact-list' | 'chat' | 'add-contact' | 'delete-contacts';

export function App() {
  const [screen, setScreen] = useState<Screen>('login');
  const [client, setClient] = useState<Foxgram | null>(null);
  const [currentContact, setCurrentContact] = useState<{
    id: string;
    username: string;
  } | null>(null);

  // Инициализация клиента
  useEffect(() => {
    const init = async () => {
      const storage = new Storage();
      const config = await storage.loadConfig();
      if (config) {
        const foxgram = new Foxgram({ serverUrl: config.serverUrl });
        setClient(foxgram);
      }
    };
    init();
  }, []);

  // Рендер текущего экрана
  const renderScreen = () => {
    switch (screen) {
      case 'login':
        return <LoginScreen onComplete={handleLogin} />;
      case 'contact-list':
        return <ContactListScreen
          onOpenChat={(id, username) => {
            setCurrentContact({ id, username });
            setScreen('chat');
          }}
          onOpenAdd={() => setScreen('add-contact')}
          onOpenDelete={() => setScreen('delete-contacts')}
          onLogout={() => setScreen('login')}
        />;
      case 'chat':
        if (!currentContact) return <LoginScreen onComplete={handleLogin} />;
        return <ChatScreen
          contactId={currentContact.id}
          contactUsername={currentContact.username}
          onBack={() => setScreen('contact-list')}
        />;
      case 'add-contact':
        return <AddContactScreen
          onBack={() => setScreen('contact-list')}
        />;
      case 'delete-contacts':
        return <DeleteContactsScreen
          onBack={() => setScreen('contact-list')}
        />;
    }
  };

  return (
    <InkApp>
      {renderScreen()}
    </InkApp>
  );
}
```

### 7.3 Обновить `src/index.ts`

```typescript
#!/usr/bin/env node

import yargs from 'yargs';
import { hideBin } from 'yargs/helpers';
import { App } from './app';
import { login } from './screens/login';
import { register } from './screens/register';
import { chat } from './screens/chat';
import { logout } from './screens/logout';

yargs(hideBin(process.argv))
  .command('login', 'Open login screen', {}, login)
  .command('register', 'Open registration screen', {}, register)
  .command('chat <username>', 'Open chat with user', (yargs) =>
    yargs.positional('username', { type: 'string', demandOption: true })
  , chat)
  .command('logout', 'Logout and clear local data', {}, logout)
  .command('*', 'Start TUI app', {}, () => {
    // По умолчанию — запустить TUI
    import('ink').then(({ App: InkApp }) => {
      import('./app').then(({ App }) => {
        InkApp(<App />);
      });
    });
  })
  .demandCommand(1, 'You need to specify a command')
  .help()
  .parse();
```
